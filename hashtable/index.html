C++ code behind the scenes

The C++ behind the scenes. Highlighted lines match the current step, which plays automatically (Find uses the same loop).

int hash(int key) { return key % 7; }
void insert(int key, string value) {
  int i = hash(key);
  Node* n = table[i];
  while (n != nullptr) {
    if (n->key == key) { n->value = value; return; }
    n = n->next;
  }
  table[i] = new Node{key, value, table[i]};
}
